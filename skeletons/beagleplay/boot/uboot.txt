#Redpesk BeaglePlay boot script commands for uboot

echo Boot into normal mode...

test -n ${consoleargs} || setenv consoleargs ttyS2,115200n8

setenv mmcroot /dev/mmcblk${devnum}p4 rootwait rw
setenv bootargs console=${consoleargs} root=${mmcroot} security=@@SECURITY@@
load mmc ${devnum}:1 ${loadaddr} Image
load mmc ${devnum}:1  ${fdt_addr_r} k3-am625-beagleplay.dtb

echo Booting with bootargs=${bootargs}
booti ${loadaddr} - ${fdt_addr_r}
