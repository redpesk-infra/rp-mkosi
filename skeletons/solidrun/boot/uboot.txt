#Redpesk Solidrun boot script commands for uboot

echo Boot into normal mode...

setenv devnum 1
setenv mmcroot /dev/mmcblk${devnum}p5 rootwait rw
setenv bootargs ${jh_clk} console=${console} root=${mmcroot} security=@@SECURITY@@
load mmc ${devnum}:2 ${kernel_addr_r} Image
load mmc ${devnum}:2 ${fdt_addr_r} imx8mp-hummingboard-iiot.dtb
booti ${kernel_addr_r} - ${fdt_addr_r}
